# FUNCTIONS
from PIL import Image
import numpy as np
import json
import random


# Internal function (not to be called directly)
def lsb_steganography_decode_word(image_path, word_number=None):
    """
    Extract a specific word or the entire message from an image
    
    Args:
        image_path (str): Path to the steganographic image
        word_number (int): Specific word number to extract (None for full message)
    
    Returns:
        str or dict: Extracted word or full message
    """
    try:
        # Open the image
        img = Image.open(image_path)
        img = img.convert('RGB')
        pixels = np.array(img)
        flat_pixels = pixels.flatten()
        
        # Extract LSB bits
        binary_data = ''
        for pixel in flat_pixels:
            binary_data += str(pixel & 1)
        
        # Convert binary to characters until null terminator
        full_text = ''
        for i in range(0, len(binary_data), 8):
            byte = binary_data[i:i+8]
            if len(byte) < 8:
                break
            char = chr(int(byte, 2))
            if char == '\x00':
                break
            full_text += char
        
        # Split into index and message
        if '\n' in full_text:
            index_part, message_part = full_text.split('\n', 1)
            
            # If specific word requested
            if word_number is not None:
                extracted_index = json.loads(index_part)
                if str(word_number) in extracted_index:
                    word_info = extracted_index[str(word_number)]
                    start = word_info['start_pos']
                    end = word_info['end_pos'] + 1
                    return message_part[start:end]
                else:
                    raise ValueError(f"Word number {word_number} not found in index")
            
            # Return just the message if no specific word requested
            return message_part
        
        # Fallback: no index found, return full text
        return full_text
        
    except Exception as e:
        print(f"Error: {e}")
        return None

def get_word_index(image_path):
    """
    Extract the word index from the steganographic image.
    Returns a dictionary where keys are word indices (as strings) and values are word information (position only).
    """
    try:
        img = Image.open(image_path)
        img = img.convert('RGB')
        pixels = np.array(img)
        flat_pixels = pixels.flatten()
        
        binary_data = ''
        for pixel in flat_pixels:
            binary_data += str(pixel & 1)
        
        text = ''
        for i in range(0, len(binary_data), 8):
            byte = binary_data[i:i+8]
            if len(byte) < 8:
                break
            char = chr(int(byte, 2))
            if char == '\x00':
                break
            text += char
        
        if '\n' in text:
            index_part, message_part = text.split('\n', 1)
            if index_part.startswith('{'):
                extracted_index = json.loads(index_part)
                # Return only positional metadata (no actual words)
                return extracted_index
        return None
    except Exception as e:
        print(f"Error extracting index: {e}")
        return None

def extract_words_by_bitmask(image_path, bitmask):
    """
    Extract specific words from the image based on a bitmask (list of word indices).
    Returns a dictionary where keys are word indices and values are the actual words.
    """
    words = {}
    for index in bitmask:
        word = lsb_steganography_decode_word(image_path, int(index))
        words[str(index)] = word
    return words

def generate_bitmask(all_indices, percentage):
    """
    Generate a random bitmask for a given percentage of words.
    all_indices: list of all word indices (as strings or integers)
    percentage: float between 0 and 1, e.g., 0.7 for 70%
    Returns a list of word indices.
    """
    num_to_select = int(len(all_indices) * percentage)
    return random.sample(all_indices, num_to_select)

def create_fill_in_blank(all_indices, revealed_words):
    """
    Create a fill-in-the-blank representation of the message.
    all_indices: list of all word indices in order (as strings)
    revealed_words: dictionary of index: word for revealed words
    Returns a string with words and blanks.
    """
    sorted_indices = sorted([int(idx) for idx in all_indices])
    sorted_indices_str = [str(idx) for idx in sorted_indices]
    parts = []
    for idx in sorted_indices_str:
        if idx in revealed_words:
            parts.append(revealed_words[idx])
        else:
            parts.append('_____')
    return ' '.join(parts)

def validate_guess(guessed_words, actual_words):
    """
    Validate guessed words against actual words.
    guessed_words: dictionary of index: guessed_word
    actual_words: dictionary of index: actual_word
    Returns a score: +1 for each correct guess, -1 for each incorrect guess.
    """
    score = 0
    for idx, guess in guessed_words.items():
        if idx in actual_words:
            if guess == actual_words[idx]:
                score += 1
            else:
                score -= 1
    return score
